# FUNÇÕES PARA CÁLCULO DOS MÉTODOS DE PREVISÃO ----------------------------

calcular_mms <- function(dados, n) {
  mms <- numeric(length(dados))
  for(i in n:length(dados)) {
    mms[i] <- mean(dados[(i-n+1):i])
  }
  return(mms)
}

calcular_mmp <- function(dados, pesos) {
  n <- length(pesos)
  mmp <- numeric(length(dados))
  for(i in n:length(dados)) {
    periodo <- dados[(i-n+1):i]
    mmp[i] <- sum(periodo * pesos) / sum(pesos)
  }
  return(mmp)
}

calcular_se <- function(dados, alpha) {
  se <- numeric(length(dados))
  se[1] <- dados[1]  # Inicialização
  for(i in 2:length(dados)) {
    se[i] <- alpha * dados[i-1] + (1-alpha) * se[i-1]
  }
  return(se)
}

calcular_mape <- function(real, previsao) {
  erro <- abs((real - previsao) / real)
  mape <- mean(erro, na.rm = TRUE) * 100
  return(round(mape, 2))
}

# FUNÇÃO PARA PÁGINA 1: DADOS E CÁLCULOS COM CAMPOS VAZIOS -----------------
criar_pagina1_calculos <- function(cenario_num, titulo, descricao, dados_reais) {
  
  # Criar dataframe com campos vazios para preenchimento
  dados_preenchimento <- data.frame(
    Mês = dados_reais$Mês,
    Demanda_Real = dados_reais$Demanda_Real,
    MMS_Aluno = rep("______", 12),
    MMP_Aluno = rep("______", 12),
    SE_Aluno = rep("______", 12),
    Erro_MMS = rep("______", 12),
    Erro_MMP = rep("______", 12),
    Erro_SE = rep("______", 12)
  )
  
  # Adicionar linha para totais e MAPE
  dados_completos <- rbind(
    dados_preenchimento,
    data.frame(
      Mês = "MAPE FINAL",
      Demanda_Real = "______",
      MMS_Aluno = "______",
      MMP_Aluno = "______",
      SE_Aluno = "______",
      Erro_MMS = "______%",
      Erro_MMP = "______%", 
      Erro_SE = "______%"
    )
  )
  
  # Criar tabela formatada com campos vazios
  tabela_calculos <- tableGrob(dados_completos, 
                              rows = NULL,
                              theme = ttheme_minimal(
                                base_size = 10,
                                core = list(
                                  fg_params = list(hjust = 0.5, x = 0.5),
                                  bg_params = list(fill = c(rep(c("white", "#f9f9f9"), 6), "#f0f8ff"))
                                ),
                                colhead = list(
                                  fg_params = list(fontface = "bold", fontsize = 11),
                                  bg_params = list(fill = "lightgray", alpha = 0.8)
                                )
                              ))
  
  # Adicionar título e instruções
  titulo_page1 <- textGrob(paste("CENÁRIO", cenario_num, ":", titulo), 
                          gp = gpar(fontsize = 16, fontface = "bold"))
  
  instrucoes <- textGrob(
    paste("INSTRUÇÕES:\n",
          "1. Calcule as previsões usando MMS (n=3), MMP (pesos: 0.5, 0.3, 0.2) e SE (α=0.3)\n",
          "2. Preencha as colunas de previsão para cada método\n", 
          "3. Calcule o erro percentual para cada período: |(Real - Previsão)/Real| × 100%\n",
          "4. Calcule o MAPE final para cada método: Média dos erros percentuais"),
    gp = gpar(fontsize = 11, lineheight = 1.2)
  )
  
   # Fórmulas para referência
   formulas <- textGrob(
     "FÓRMULAS: MMS = (Dₜ + Dₜ₋₁ + Dₜ₋₂)/3  |  MMP = (0.5×Dₜ + 0.3×Dₜ₋₁ + 0.2×Dₜ₋₂)  |  SE = α×Dₜ + (1-α)×Fₜ",
     gp = gpar(fontsize = 10, fontface = "italic")
   )
  
  # Layout da página 1
  grid.arrange(
    titulo_page1,
    instrucoes,
    formulas,
    tabela_calculos,
    nrow = 4,
    heights = c(0.3, 0.4, 0.2, 3)
  )
}

# FUNÇÃO PARA PÁGINA 2: GRÁFICO DE PREVISÕES ------------------------------
criar_pagina2_grafico <- function(cenario_num, titulo, dados_reais) {
  
  # Criar gráfico com eixos e pontos da demanda real
  grafico_previsoes <- ggplot() +
    # Grade de fundo
    theme_bw() +
    
    # Eixos e limites
    scale_x_continuous(breaks = 1:12, labels = dados_reais$Mês, expand = c(0.02, 0.02)) +
    scale_y_continuous(limits = c(0, max(dados_reais$Demanda_Real) * 1.4)) +
    
    # Pontos da demanda real
    geom_point(aes(x = 1:12, y = dados_reais$Demanda_Real), 
               size = 3, color = "black", shape = 16) +
    
    # Linha da demanda real
    geom_line(aes(x = 1:12, y = dados_reais$Demanda_Real), 
              color = "black", size = 1, linetype = "solid") +
    
    # Áreas para desenhar as previsões (linhas tracejadas como guia)
    geom_hline(yintercept = seq(0, max(dados_reais$Demanda_Real) * 1.3, 
                               length.out = 10), 
               color = "gray90", size = 0.2) +
    
    # Títulos e labels
    labs(
      title = paste("CENÁRIO", cenario_num, ":", titulo, "- REGISTRO DAS PREVISÕES"),
      subtitle = "Demanda Real (linha preta) - Desenhe as previsões usando as cores indicadas",
      x = "Meses",
      y = "Demanda"
    ) +
    
    # Tema
    theme(
      plot.title = element_text(hjust = 0.5, face = "bold", size = 14),
      plot.subtitle = element_text(hjust = 0.5, size = 11),
      panel.grid.major = element_line(color = "gray95"),
      panel.grid.minor = element_blank()
    ) +
    
    # Legenda interativa para cores
    annotate("text", x = 1, y = max(dados_reais$Demanda_Real) * 1.35, 
             label = "LEGENDA DE CORES:", hjust = 0, fontface = "bold", size = 4) +
    annotate("text", x = 1, y = max(dados_reais$Demanda_Real) * 1.30, 
             label = "● MMS (Média Móvel Simples) → Use VERMELHO", hjust = 0, size = 3.5) +
    annotate("text", x = 1, y = max(dados_reais$Demanda_Real) * 1.26, 
             label = "● MMP (Média Móvel Ponderada) → Use AZUL", hjust = 0, size = 3.5) +
    annotate("text", x = 1, y = max(dados_reais$Demanda_Real) * 1.22, 
             label = "● SE (Suavização Exponencial) → Use VERDE", hjust = 0, size = 3.5) +
    
    # Espaço para anotações
    annotate("text", x = 8, y = max(dados_reais$Demanda_Real) * 1.30, 
             label = "OBSERVAÇÕES:", hjust = 0, fontface = "bold", size = 4) +
    annotate("text", x = 8, y = max(dados_reais$Demanda_Real) * 1.26, 
             label = "• Qual método acompanhou melhor a demanda?", hjust = 0, size = 3) +
    annotate("text", x = 8, y = max(dados_reais$Demanda_Real) * 1.20, 
             label = "• Justifique sua resposta:", hjust = 0, size = 3)
  
  return(grafico_previsoes)
}


# VERSÃO COMBINADA (DUAS PÁGINAS EM UM PDF) -------------------------------

criar_pdf_completo <- function(cenario_num, titulo, dados_reais, nome_arquivo) {
  
  pdf(nome_arquivo, width = 11, height = 8.5)
  
  # Página 1 - Cálculos
  criar_pagina1_calculos(cenario_num, titulo, "", dados_reais)
  
  # Página 2 - Gráfico
  print(criar_pagina2_grafico(cenario_num, titulo, dados_reais))
  
  dev.off()
}

# CALCULAR RESULTADOS CORRETOS PARA O GABARITO ----------------------------

# Função para calcular resultados corretos
calcular_resultados_gabarito <- function(dados_reais) {
  
  # Calcular previsões corretas
  mms <- calcular_mms(dados_reais$Demanda_Real, 3)
  mmp <- calcular_mmp(dados_reais$Demanda_Real, c(0.5, 0.3, 0.2))
  se <- calcular_se(dados_reais$Demanda_Real, 0.3)
  
  # Calcular MAPE corretos
  mape_mms <- calcular_mape(dados_reais$Demanda_Real[4:12], mms[4:12])
  mape_mmp <- calcular_mape(dados_reais$Demanda_Real[4:12], mmp[4:12])
  mape_se <- calcular_mape(dados_reais$Demanda_Real[2:12], se[2:12])
  
  return(list(
    MMS = mms,
    MMP = mmp,
    SE = se,
    MAPE_MMS = mape_mms,
    MAPE_MMP = mape_mmp,
    MAPE_SE = mape_se
  ))
}

# Calcular resultados para cada cenário
resultados1 <- calcular_resultados_gabarito(df1)
resultados2 <- calcular_resultados_gabarito(df2)
resultados3 <- calcular_resultados_gabarito(df3)

# FUNÇÃO CORRIGIDA DO GABARITO --------------------------------------------

criar_gabarito <- function(cenario_num, titulo, dados_reais, resultados) {
  
  pdf(paste0("Gabarito_Cenario", cenario_num, ".pdf"), width = 11, height = 8.5)
  
  # Criar dataframe completo para o gabarito
  dados_gabarito <- data.frame(
    Mês = dados_reais$Mês,
    Demanda_Real = dados_reais$Demanda_Real,
    MMS_Correto = round(resultados$MMS, 1),
    MMP_Correto = round(resultados$MMP, 1),
    SE_Correto = round(resultados$SE, 1),
    Erro_MMS = round(abs((dados_reais$Demanda_Real - resultados$MMS) / dados_reais$Demanda_Real) * 100, 1),
    Erro_MMP = round(abs((dados_reais$Demanda_Real - resultados$MMP) / dados_reais$Demanda_Real) * 100, 1),
    Erro_SE = round(abs((dados_reais$Demanda_Real - resultados$SE) / dados_reais$Demanda_Real) * 100, 1)
  )
  
  # Substituir NA por "-"
  dados_gabarito[is.na(dados_gabarito)] <- "-"
  
  # Adicionar linha de MAPE
  linha_mape <- data.frame(
    Mês = "MAPE FINAL",
    Demanda_Real = "-",
    MMS_Correto = paste(resultados$MAPE_MMS, "%"),
    MMP_Correto = paste(resultados$MAPE_MMP, "%"),
    SE_Correto = paste(resultados$MAPE_SE, "%"),
    Erro_MMS = "-",
    Erro_MMP = "-",
    Erro_SE = "-"
  )
  
  dados_gabarito_completo <- rbind(dados_gabarito, linha_mape)
  
  # Criar tabela formatada
  tabela_gabarito <- tableGrob(dados_gabarito_completo, rows = NULL,
                              theme = ttheme_minimal(
                                base_size = 9,
                                core = list(
                                  fg_params = list(hjust = 0.5, x = 0.5),
                                  bg_params = list(fill = c(rep(c("white", "#f0f0f0"), 6), "#e8f5e8"))
                                ),
                                colhead = list(
                                  fg_params = list(fontface = "bold", fontsize = 10),
                                  bg_params = list(fill = "#2196F3", alpha = 0.8)
                                )
                              ))
  
  # Layout da página do gabarito
  grid.arrange(
    textGrob(paste("GABARITO - CENÁRIO", cenario_num, ":", titulo),
            gp = gpar(fontsize = 16, fontface = "bold")),
    
    textGrob("RESPOSTAS CORRETAS PARA OS CÁLCULOS DE PREVISÃO",
            gp = gpar(fontsize = 12, fontface = "italic")),
    
    textGrob(paste("MAPE Esperado: MMS =", resultados$MAPE_MMS, "%, MMP =", 
                  resultados$MAPE_MMP, "%, SE =", resultados$MAPE_SE, "%"),
            gp = gpar(fontsize = 11, col = "darkred")),
    
    tabela_gabarito,
    
    nrow = 4, 
    heights = c(0.2, 0.1, 0.1, 2.5)
  )
  
  dev.off()
}

# GABARITO GRÁFICO COM PREVISÕES PLOTADAS ---------------------------------

criar_gabarito_grafico <- function(cenario_num, titulo, dados_reais, resultados) {
  
  pdf(paste0("Gabarito_Grafico_Cenario", cenario_num, ".pdf"), width = 11, height = 8.5)
  
  # Criar gráfico com todas as previsões
  grafico_gabarito <- ggplot() +
    # Demanda real
    geom_point(aes(x = 1:12, y = dados_reais$Demanda_Real), 
               size = 3, color = "black", shape = 16) +
    geom_line(aes(x = 1:12, y = dados_reais$Demanda_Real), 
              color = "black", size = 1, linetype = "solid") +
    
    # Previsão MMS
    geom_line(aes(x = 4:12, y = resultados$MMS[4:12]), 
              color = "red", size = 1, linetype = "dashed") +
    geom_point(aes(x = 4:12, y = resultados$MMS[4:12]), 
               size = 2, color = "red", shape = 17) +
    
    # Previsão MMP
    geom_line(aes(x = 4:12, y = resultados$MMP[4:12]), 
              color = "blue", size = 1, linetype = "dashed") +
    geom_point(aes(x = 4:12, y = resultados$MMP[4:12]), 
               size = 2, color = "blue", shape = 15) +
    
    # Previsão SE
    geom_line(aes(x = 2:12, y = resultados$SE[2:12]), 
              color = "darkgreen", size = 1, linetype = "dashed") +
    geom_point(aes(x = 2:12, y = resultados$SE[2:12]), 
               size = 2, color = "darkgreen", shape = 18) +
    
    # Configurações do gráfico
    scale_x_continuous(breaks = 1:12, labels = dados_reais$Mês, expand = c(0.02, 0.02)) +
    scale_y_continuous(limits = c(0, max(dados_reais$Demanda_Real) * 1.3)) +
    
    labs(
      title = paste("GABARITO - CENÁRIO", cenario_num, ":", titulo),
      subtitle = "Linha preta: Demanda Real | Vermelho: MMS | Azul: MMP | Verde: SE",
      x = "Meses",
      y = "Demanda"
    ) +
    
    theme_bw() +
    theme(
      plot.title = element_text(hjust = 0.5, face = "bold", size = 14),
      plot.subtitle = element_text(hjust = 0.5, size = 11),
      panel.grid.major = element_line(color = "gray95")
    ) +
    
    # Legenda
    annotate("text", x = 1, y = max(dados_reais$Demanda_Real) * 1.25, 
             label = paste("MAPE - MMS:", resultados$MAPE_MMS, "%"), 
             hjust = 0, color = "red", fontface = "bold") +
    annotate("text", x = 1, y = max(dados_reais$Demanda_Real) * 1.20, 
             label = paste("MAPE - MMP:", resultados$MAPE_MMP, "%"), 
             hjust = 0, color = "blue", fontface = "bold") +
    annotate("text", x = 1, y = max(dados_reais$Demanda_Real) * 1.15, 
             label = paste("MAPE - SE:", resultados$MAPE_SE, "%"), 
             hjust = 0, color = "darkgreen", fontface = "bold")
  
  print(grafico_gabarito)
  dev.off()
}

# RESUMO COMPARATIVO PARA O PROFESSOR ------------------------------------

criar_resumo_professor <- function() {
  
  resumo <- data.frame(
    Cenário = c("1 - Sorvete Artesanal", "2 - Smartphone Premium", "3 - Medicamento Específico"),
    Padrão = c("Sazonal", "Tendência", "Irregular"),
    MAPE_MMS = c(resultados1$MAPE_MMS, resultados2$MAPE_MMS, resultados3$MAPE_MMS),
    MAPE_MMP = c(resultados1$MAPE_MMP, resultados2$MAPE_MMP, resultados3$MAPE_MMP),
    MAPE_SE = c(resultados1$MAPE_SE, resultados2$MAPE_SE, resultados3$MAPE_SE),
    Melhor_Método = c(
      ifelse(min(c(resultados1$MAPE_MMS, resultados1$MAPE_MMP, resultados1$MAPE_SE)) == resultados1$MAPE_MMS, "MMS",
             ifelse(min(c(resultados1$MAPE_MMS, resultados1$MAPE_MMP, resultados1$MAPE_SE)) == resultados1$MAPE_MMP, "MMP", "SE")),
      ifelse(min(c(resultados2$MAPE_MMS, resultados2$MAPE_MMP, resultados2$MAPE_SE)) == resultados2$MAPE_MMS, "MMS",
             ifelse(min(c(resultados2$MAPE_MMS, resultados2$MAPE_MMP, resultados2$MAPE_SE)) == resultados2$MAPE_MMP, "MMP", "SE")),
      ifelse(min(c(resultados3$MAPE_MMS, resultados3$MAPE_MMP, resultados3$MAPE_SE)) == resultados3$MAPE_MMS, "MMS",
             ifelse(min(c(resultados3$MAPE_MMS, resultados3$MAPE_MMP, resultados3$MAPE_SE)) == resultados3$MAPE_MMP, "MMP", "SE"))
    )
  )
  
  pdf("Resumo_Professor.pdf", width = 11, height = 8.5)
  
  tabela_resumo <- tableGrob(resumo, rows = NULL,
                            theme = ttheme_minimal(
                              base_size = 11,
                              core = list(
                                bg_params = list(fill = c(rep(c("white", "#f9f9f9"), 2), "white"))
                              ),
                              colhead = list(
                                fg_params = list(fontface = "bold"),
                                bg_params = list(fill = "#FF9800")
                              )
                            ))
  
  grid.arrange(
    textGrob("RESUMO PARA O PROFESSOR - ANÁLISE COMPARATIVA",
            gp = gpar(fontsize = 16, fontface = "bold")),
    
    textGrob("Comparação do MAPE por Cenário e Método de Previsão",
            gp = gpar(fontsize = 12, fontface = "italic")),
    
    tabela_resumo,
    
    textGrob("ANÁLISE:\n• Cenário 1 (Sazonal): Método X funciona melhor porque...\n• Cenário 2 (Tendência): Método Y é mais adequado porque...\n• Cenário 3 (Irregular): Método Z se adapta melhor porque...",
            gp = gpar(fontsize = 10, lineheight = 1.2),
            x = unit(0.05, "npc"), just = "left"),
    
    nrow = 4,
    heights = c(0.2, 0.1, 0.6, 0.4)
  )
  
  dev.off()
  
  return(resumo)
}

# Gerar resumo para o professor
resumo_professor <- criar_resumo_professor()

print(resumo_professor)
