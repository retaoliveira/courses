2. CÁLCULO DAS PREVISÕES - MÉTODO DA MÉDIA MÓVEL SIMPLES (n=3)

=============================================================================

calcular_mms <- function(dados, n = 3) {
  previsoes <- rep(NA, nrow(dados))
  for (i in (n+1):nrow(dados)) {
    previsoes[i] <- mean(dados$Demanda_Real[(i-n):(i-1)])
  }
  # Previsão para o próximo período (Mês 13)
  previsao_futura <- mean(tail(dados$Demanda_Real, n))
  return(list(previsoes = previsoes, proxima_previsao = previsao_futura))
}

resultado_mms <- calcular_mms(dados_demanda)
dados_demanda$MMS <- resultado_mms$previsoes

=============================================================================

calcular_mmp <- function(dados, pesos = c(0.5, 0.3, 0.2)) {
  n <- length(pesos)
  previsoes <- rep(NA, nrow(dados))
  
  for (i in (n+1):nrow(dados)) {
    valores <- dados$Demanda_Real[(i-n):(i-1)]
    previsoes[i] <- sum(valores * pesos)
  }
  
  # Previsão para o próximo período (Mês 13)
  previsao_futura <- sum(tail(dados$Demanda_Real, n) * pesos)
  return(list(previsoes = previsoes, proxima_previsao = previsao_futura))
}

resultado_mmp <- calcular_mmp(dados_demanda)
dados_demanda$MMP <- resultado_mmp$previsoes

4. CÁLCULO DAS PREVISÕES - SUAVIZAÇÃO EXPONENCIAL (α = 0.3)

=============================================================================

calcular_ses <- function(dados, alpha = 0.3) {
  previsoes <- numeric(nrow(dados))
  previsoes[1] <- dados$Demanda_Real[1]  # Previsão inicial = primeira demanda
  
  for (i in 2:nrow(dados)) {
    previsoes[i] <- alpha * dados$Demanda_Real[i-1] + (1 - alpha) * previsoes[i-1]
  }
  
  # Previsão para o próximo período (Mês 13)
  previsao_futura <- alpha * tail(dados$Demanda_Real, 1) + (1 - alpha) * tail(previsoes, 1)
  return(list(previsoes = previsoes, proxima_previsao = previsao_futura))
}

resultado_ses <- calcular_ses(dados_demanda)
dados_demanda$SES <- resultado_ses$previsoes

5. CÁLCULO DOS ERROS DE PREVISÃO (MAPE)

=============================================================================

calcular_mape <- function(real, previsao) {
  # Remover NAs para cálculo
  indices_validos <- !is.na(previsao)
  real_validos <- real[indices_validos]
  previsao_validos <- previsao[indices_validos]
  
  erro_percentual <- abs((real_validos - previsao_validos) / real_validos) * 100
  mape <- mean(erro_percentual)
  return(mape)
}

# Calcular MAPE para cada método
mape_mms <- calcular_mape(dados_demanda$Demanda_Real, dados_demanda$MMS)
mape_mmp <- calcular_mape(dados_demanda$Demanda_Real, dados_demanda$MMP)
mape_ses <- calcular_mape(dados_demanda$Demanda_Real, dados_demanda$SES)
