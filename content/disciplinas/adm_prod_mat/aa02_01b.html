💻 Códigos Principais Implementados

1. Função de Análise de Padrões

analisar_padrao_demanda <- function(dados) {
  modelo <- lm(demanda ~ mes, data = dados)
  tendencia <- coefficients(modelo)[2]
  r2 <- summary(modelo)$r.squared
  cv <- sd(dados$demanda) / mean(dados$demanda) * 100
  
  tipo_padrao <- ifelse(cv > 40, "Alta Volatilidade",
                       ifelse(r2 > 0.8, "Tendência Forte", 
                             ifelse(abs(tendencia) > 10, "Crescimento", "Sazonal")))
  
  return(list(tendencia = tendencia, r2 = r2, cv = cv, tipo = tipo_padrao))
}

2. Funções de Previsão por Cenário

# CENÁRIO 1 - SAZONALIDADE
prever_sazonal <- function(dados) {
  # Decomposição sazonal simples
  media_verao <- mean(dados$demanda[5:8])  # Meses 5-8
  media_inverno <- mean(dados$demanda[10:12]) # Meses 10-12
  return(list(
    previsao_mes13 = media_inverno,  # Mês 13 é inverno
    tecnica = "Média Sazonal",
    intervalo = c(media_inverno*0.9, media_inverno*1.1)
  ))
}

# CENÁRIO 2 - CRESCIMENTO  
prever_crescimento <- function(dados) {
  modelo <- lm(Demanda_Real ~ Mes, data = dados)
  previsao <- predict(modelo, newdata = data.frame(Mes = 13))
  return(list(
    previsao_mes13 = round(previsao, 1),
    tecnica = "Regressão Linear",
    r2 = summary(modelo)$r.squared
  ))
}

# CENÁRIO 3 - VOLATILIDADE
prever_volatil <- function(dados) {
  mms <- mean(tail(dados$Demanda_Real, 3))  # MMS n=3
  return(list(
    previsao_mes13 = mms,
    tecnica = "Média Móvel Simples (n=3)",
    estoque_seguranca = sd(dados$Demanda_Real) * 1.5
  ))
}

3. Geração de Relatórios Automatizados

gerar_relatorio_cenario <- function(cenario, numero) {
  analise <- analisar_padrao_demanda(cenario$dados)
  
  cat("=== CENÁRIO", numero, "===\n")
  cat("Produto:", cenario$produto, "\n")
  cat("Padrão:", analise$tipo, "\n")
  cat("CV:", round(analise$cv, 1), "% | Tendência:", round(analise$tendencia, 2), "/mês\n")
  
  # Aplicar técnica recomendada
  if(analise$tipo == "Sazonal") {
    resultado <- prever_sazonal(cenario$dados)
  } else if(analise$tipo == "Crescimento") {
    resultado <- prever_crescimento(cenario$dados)  
  } else {
    resultado <- prever_volatil(cenario$dados)
  }
  
  cat("Técnica:", resultado$tecnica, "\n")
  cat("Previsão Mês 13:", resultado$previsao_mes13, "unidades\n")
}
